cmake_minimum_required(VERSION 3.16)
project(wcis_lan_stream LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
add_library(wcis_jpeg_publisher INTERFACE)
target_include_directories(wcis_jpeg_publisher INTERFACE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(wcis_jpeg_publisher INTERFACE ${OpenCV_LIBS} Threads::Threads)
add_executable(wcis_stream_demo src/stream_demo.cpp)
target_link_libraries(wcis_stream_demo PRIVATE wcis_jpeg_publisher)
target_compile_options(wcis_stream_demo PRIVATE -Wall -Wextra -Werror)
add_executable(wcis_raw_demo src/raw_demo.cpp)
target_include_directories(wcis_raw_demo PRIVATE include)
target_link_libraries(wcis_raw_demo PRIVATE Threads::Threads)
target_compile_options(wcis_raw_demo PRIVATE -Wall -Wextra -Werror)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../ps_app/example_include/wcis_capture_wait.hpp")
  add_executable(wcis_capture_wait_test src/capture_wait_test.cpp)
  target_include_directories(wcis_capture_wait_test PRIVATE ../ps_app/example_include)
  target_compile_options(wcis_capture_wait_test PRIVATE -Wall -Wextra -Werror)
endif()
