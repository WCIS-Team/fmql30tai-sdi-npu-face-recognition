cmake_minimum_required(VERSION 3.16)
project(wcis_sdi_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The vendor logging macros use GNU's empty variadic-argument extension.
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_SKIP_RPATH ON)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "This app requires the aarch64 on-chip SDK or a matching cross toolchain")
endif()
find_package(icraft-zg330backend 3.39.0 REQUIRED)
find_package(icraft-hostbackend 3.39.0 REQUIRED)
find_package(icraft-buyibackend 3.39.0 REQUIRED)
set(MODELZOO_STATIC /ModelzooDeps/aarch64/Static CACHE PATH "Vendor third-party static SDK")
set(MODELZOO_DYNAMIC /ModelzooDeps/aarch64/Dynamic CACHE PATH "Vendor third-party dynamic SDK")
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs
  PATHS ${MODELZOO_STATIC}/lib/cmake/opencv4)
find_package(yaml-cpp REQUIRED PATHS ${MODELZOO_STATIC}/lib/cmake/yaml-cpp
  ${MODELZOO_DYNAMIC}/lib/cmake/yaml-cpp)
option(WCIS_BUILD_REFERENCE "Build the existing YOLO reference" ON)
if(WCIS_BUILD_REFERENCE)
  add_executable(wcis_sdi_reference src/wcis_sdi_reference.cpp)
endif()
add_executable(wcis_camera_preview src/wcis_camera_preview.cpp)
add_executable(wcis_camera_single_frame src/wcis_camera_single_frame.cpp)
add_executable(wcis_sdi_vpu_live src/wcis_sdi_vpu_live.cpp)
target_include_directories(wcis_sdi_vpu_live PRIVATE ../npu_photo/vendor)
option(WCIS_BUILD_CAMERA_OWNER "Build the single-owner camera/face backend" OFF)
set(WCIS_TARGETS wcis_camera_preview wcis_camera_single_frame wcis_sdi_vpu_live)
if(WCIS_BUILD_CAMERA_OWNER)
  add_executable(wcis_camera_owner src/wcis_camera_owner.cpp)
  target_include_directories(wcis_camera_owner PRIVATE ../npu_photo/vendor ../npu_photo/src)
  list(APPEND WCIS_TARGETS wcis_camera_owner)
endif()
option(WCIS_BUILD_ENCODER_TRIAL "Build the bounded real-SDI PL/VPU encoder comparison" OFF)
option(WCIS_BUILD_PL_NPU_TRIAL "Build the bounded two-frame FAC1/NPU overlap probe" OFF)
if(WCIS_BUILD_PL_NPU_TRIAL)
  add_executable(wcis_npu_pl_trial src/wcis_npu_pl_trial.cpp)
  target_include_directories(wcis_npu_pl_trial PRIVATE ../npu_photo/vendor ../npu_photo/src)
  list(APPEND WCIS_TARGETS wcis_npu_pl_trial)
endif()
if(WCIS_BUILD_ENCODER_TRIAL)
  add_executable(wcis_encoder_trial src/wcis_encoder_trial.cpp)
  target_include_directories(wcis_encoder_trial PRIVATE ../npu_photo/vendor ../npu_photo/src)
  list(APPEND WCIS_TARGETS wcis_encoder_trial)
endif()
if(WCIS_BUILD_REFERENCE)
  list(APPEND WCIS_TARGETS wcis_sdi_reference)
endif()
foreach(WCIS_TARGET IN LISTS WCIS_TARGETS)
target_compile_definitions(${WCIS_TARGET} PRIVATE USE_ZG330_BACKEND)
target_compile_options(${WCIS_TARGET} PRIVATE -fmax-errors=5)
target_include_directories(${WCIS_TARGET} PRIVATE ../lan_stream/include
  example_include vendor/include ${MODELZOO_STATIC}/include
  ${MODELZOO_DYNAMIC}/include ${MODELZOO_STATIC}/include/opencv4
  ${MODELZOO_STATIC}/include/Eigen-3.4.0
  ${MODELZOO_STATIC}/include/Eigen-3.4.0/Eigen
  ${MODELZOO_STATIC}/include/OpenEXR ${MODELZOO_STATIC}/include/ffmpeg)
set(WCIS_UAPI_ROOT "" CACHE PATH "Matching board Linux UAPI headers for cross compilation")
if(WCIS_UAPI_ROOT)
  target_include_directories(${WCIS_TARGET} SYSTEM PRIVATE
    ${WCIS_UAPI_ROOT}/usr/include
    ${WCIS_UAPI_ROOT}/usr/include/aarch64-linux-gnu)
endif()
target_link_directories(${WCIS_TARGET} PRIVATE
  ${MODELZOO_STATIC}/lib ${MODELZOO_DYNAMIC}/lib)
target_link_libraries(${WCIS_TARGET} PRIVATE
  Icraft::ZG330Backend Icraft::HostBackend Icraft::BuyiBackend
  ${OpenCV_LIBS} yaml-cpp pthread dl atomic stdc++fs)
endforeach()
add_executable(wcis_capture_wait_test ../lan_stream/src/capture_wait_test.cpp)
target_include_directories(wcis_capture_wait_test PRIVATE example_include)
target_link_libraries(wcis_capture_wait_test PRIVATE pthread)
target_compile_options(wcis_capture_wait_test PRIVATE -Wall -Wextra -Werror)
