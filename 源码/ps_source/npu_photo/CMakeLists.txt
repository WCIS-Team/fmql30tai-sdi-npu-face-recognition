cmake_minimum_required(VERSION 3.16)
project(wcis_face_npu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SKIP_RPATH ON)
find_package(icraft-zg330backend 3.39.0 REQUIRED)
find_package(icraft-hostbackend 3.39.0 REQUIRED)
add_executable(wcis_pl_net_config src/pl_net_config.cpp)
target_include_directories(wcis_pl_net_config PRIVATE vendor)
if(WCIS_UAPI_ROOT)
  target_include_directories(wcis_pl_net_config SYSTEM PRIVATE ${WCIS_UAPI_ROOT}/usr/include ${WCIS_UAPI_ROOT}/usr/include/aarch64-linux-gnu)
endif()
target_compile_options(wcis_pl_net_config PRIVATE -Wall -Wextra -fmax-errors=5)
target_link_libraries(wcis_pl_net_config PRIVATE Icraft::ZG330Backend pthread dl atomic stdc++fs)
